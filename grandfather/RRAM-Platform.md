### manifold/transfer.ts

```typescript
// Enforces: Phi_{n+1} = Psi_n
// Every layer transition must pass through this gate

export type TransferResult =
  | { ok: true;  phi_next: IntentField }
  | { ok: false; reason: string; blocked_at: LayerIndex };

export function enforceStateTransfer(
  psi_n: StateField,
  layer_n: LayerIndex
): TransferResult {
  const layer_next = (layer_n + 1) as LayerIndex;

  // Psi_n must be stable before Phi_{n+1} can be defined
  if (!isStable(psi_n)) {
    return {
      ok: false,
      reason: `Layer ${layer_n} (${LAYERS[layer_n].name}) is not stable. 
               Phi_${layer_next} is undefined.`,
      blocked_at: layer_n,
    };
  }

  // Phi_{n+1} IS Psi_n — not derived from it, identical to it
  return {
    ok: true,
    phi_next: {
      layer: layer_next,
      coordinate: psi_n.coordinate,
      declared_at: new Date(),
      phi: psi_n.psi,  // identity, not derivation
    },
  };
}

function isStable(psi: StateField): boolean {
  // A state is stable when it has been realized and verified
  // Each layer overrides this with its specific stability condition
  return Object.keys(psi.psi).length > 0;
}
```

```typescript
const SCHEMA_CUSTOMER_PAGES = `

      created_at  TIMESTAMPTZ DEFAULT NOW(),
      updated_at  TIMESTAMPTZ DEFAULT NOW()

  // Called once at server init (hooks.server.ts → init())
  // Creates all structural tables + seeds structural pages
  // This is Phi_2 materializing Psi_2 — one atomic operation

  await query(SCHEMA_CUSTOMER_PAGES);

  // Also creates supporting tables (messages, bookings, documents)
  // each with user_id FK → schema closure = G co-located with Phi
  await query(SCHEMA_MESSAGES);
  await query(SCHEMA_BOOKINGS);
  await query(SCHEMA_DOCUMENTS);

  for (const page of SEED_PAGES) {
    // Create the user_data table for each seed page
    await query(userDataTableSql(page.slug));

    // Insert the page row if not exists
    await query(
      `INSERT INTO customer_pages (label, slug, route_path, status, sort_order)
       VALUES ($1,$2,$3,'live',$4) ON CONFLICT (slug) DO NOTHING`,
      [page.label, page.slug, page.route_path, page.sort_order]

    // Seed the default body if not yet set
    const defaultBody = getDefaultBody(page.slug);
    if (defaultBody) {
      await query(
        `UPDATE customer_pages SET body=$1 WHERE slug=$2 AND body IS NULL`,
        [defaultBody, page.slug]
      );
    }
  }
}
```

### layer_3/render.ts — T_x: Single Operator at Two Coordinates

```typescript
// The execution operator T parameterized by coordinate x
// Delta_2: customer view (x = self, user_id = i_0)
// Delta_3: admin observer view (x = observer, target_id = customer_id)
// SAME function. DIFFERENT coordinate. ZERO translation cost.

export type RenderContext =
  | { mode: 'customer'; user_id: string }
  | { mode: 'admin';    user_id: string; target_id: string };

export async function renderPage(
  slug: string,
  context: RenderContext
): Promise {
  // Fetch Phi_2 (the template)
  const page = await queryOne(
    `SELECT label, body, status FROM customer_pages WHERE slug=$1`,
    [slug]
  );

  if (!page || page.status !== 'live') return { label: '', body: null };

  // The dipole: inject coordinate into body if admin mode
  // Customer mode: body renders as-is (window.__ADMIN_CUSTOMER_ID is absent)
  // Admin mode: body gets the target coordinate injected
  const body = context.mode === 'admin'
    ? injectAdminCoordinate(page.body, context.target_id)
    : page.body;

  // Measure Delta-Psi_3 after render
  await measureDeltaPsi3(slug, context);

  return { label: page.label, body };
}

function injectAdminCoordinate(body: string | null, target_id: string): string | null {
  if (!body) return null;
  // Injects window.__ADMIN_CUSTOMER_ID before 
  // This is the coordinate parameterization that makes admin = observer
  const injection = `
    window.__ADMIN_CUSTOMER_ID = '${target_id}';
  `;
  return body.replace('', `${injection}`);
}
```

### layer_4/utility.ts — U(x,t): User Utility Field

```typescript
// U(x,t) = 1 - normalized(|Delta-Psi_3(x,t)|)
// U = 1: perfect experience, zero friction
// U = 0: complete failure, user cannot complete intent
// This is the LEADING indicator — revenue is lagging

export type UtilityReading = {

  slug: string;          // which page
  u: number;             // [0,1]
  measured_at: Date;
};

export async function computeUtility(
  user_id: string,
  slug: string,
  window_hours = 24
): Promise {
  const recent = await query(
    `SELECT delta_psi_3 FROM behavior_log
     WHERE user_id=$1 AND slug=$2
       AND measured_at > NOW() - INTERVAL '${window_hours} hours'
     ORDER BY measured_at DESC LIMIT 50`,
    [user_id, slug]

  const avg_delta = recent.length
    ? recent.reduce((s, r) => s + r.delta_psi_3, 0) / recent.length
    : 0;

  return {
```

### layer_5/locks.ts — The Four Constraint Locks

```typescript
// Hard invariants — the immune system of the profit miner
// No mutation passes without satisfying all four locks
// These are NOT guidelines. They are G(Phi_5, x) at the AI layer.

export type LockResult =
  | { passed: true;  mutation: Mutation }
  | { passed: false; lock: 1 | 2 | 3 | 4; reason: string; mutation: Mutation };

export async function applyConstraintLocks(
  proposed: OperatorDecision,
  a_before: ActionReading
): Promise {
  const locked_mutations: Mutation[] = [];

  for (const mutation of proposed.mutations) {
    const result = await checkAllLocks(mutation, a_before);
    if (result.passed) {
      locked_mutations.push(mutation);
    } else {
      // Lock failed — log to memory field as G_event, do not surface mutation
      await appendToMemory({
        layer: mutation.layer,
        type: 'gate_event',
        value: { fired: false, lock: result.lock, reason: result.reason, mutation },
      });
    }
  }

  return { ...proposed, mutations: locked_mutations };
}

async function checkAllLocks(
  mutation: Mutation,
  a_before: ActionReading
): Promise {
  // Lock 1: Persistence Gate Invariant
  // S_n_quality(after M) >= S_n_quality(before M) for all n
  const s_before = await computePerLayerPersistence();
  const s_predicted = predictPersistenceAfter(mutation, s_before);
  for (const [n, s] of Object.entries(s_predicted)) {
    if (s < s_before[n as unknown as LayerIndex]) {
      return { passed: false, lock: 1, reason: `Layer ${n} persistence would decrease`, mutation };
    }
  }

  // Lock 2: Mismatch Monotonicity
  // |Delta-Psi_n(after M)| <= |Delta-Psi_n(before M)| for all n
  if (mutation.predicted_delta_psi_reduction < 0) {
    return { passed: false, lock: 2, reason: 'Mutation would increase mismatch', mutation };
  }

  // Lock 3: Temporal Stagger Requirement
  // For back-edge mutations: tau_{n-k} >> tau_n
  if (isBackEdge(mutation)) {
    const target_tau = LAYERS[mutation.layer].timescale_ms;
    const writer_tau = LAYERS[getWriterLayer(mutation)].timescale_ms;
    if (target_tau <= writer_tau * 10) {
      return { passed: false, lock: 3, reason: 'Back-edge timescale ratio insufficient', mutation };
    }
  }

  // Lock 4: Human Override
  // All Phi declarations require human approval
  // This is epistemological necessity: the AI's Phi model is an approximation
  // The human is the ground truth of Phi_0
  if (isPhiDeclaration(mutation)) {
    return {
      passed: true,
      mutation: { ...mutation, requires_human_approval: true }
    };
  }

  return { passed: true, mutation };
}
```

```typescript
// Psi_n[t+1] = Psi_n[t] + T_x(Phi_n - Psi_n[t]) * G(Phi_n, x)

    measureLayer0(),
    measureLayer1Summary(),
    measureLayer2(),
    measureLayer3(),
    measureLayer4(),

  const decision = await proposeArchitecturalMutation({
    0: l0.delta_psi,

    2: l2.delta_psi,
    3: l3.delta_psi,
    4: l4.delta_psi,
    5: 0,  // layer 5 self-reports as stable when running

  // Auto-approved mutations: apply immediately (only Psi changes, never Phi)

      await queueForApproval(mutation);
    } else {

    }
  }

  // Shorter cycles during high-gradient periods (drift accelerating)
```
